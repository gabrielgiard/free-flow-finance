name: Update prices and rebuild site

# Runs automatically on a schedule, and whenever you click "Run workflow"
# in the Actions tab. Markets are closed at weekends, so this only runs on
# weekdays.
on:
  schedule:
    # 22:30 UTC Mon-Fri = ~6:30pm New York, after the US close.
    # For every-other-day instead, change MON-FRI to MON,WED,FRI
    - cron: "30 22 * * MON-FRI"
  workflow_dispatch:        # lets you trigger it manually from the Actions tab

# The job commits the regenerated data file back to the repo, so it needs
# write access to repository contents.
permissions:
  contents: write

# If a run is already going, don't start a second one on top of it.
concurrency:
  group: update-prices
  cancel-in-progress: false

jobs:
  update:
    runs-on: ubuntu-latest
    steps:
      - name: Check out the repository
        uses: actions/checkout@v5

      - name: Set up Python
        uses: actions/setup-python@v6
        with:
          python-version: "3.12"

      # Gather all the data first, then build the site from it. Order matters:
      # build.py reads prices.json and market.json, so both fetches must
      # complete before it runs.
      - name: Fetch live prices from Finnhub
        env:
          FINNHUB_API_KEY: ${{ secrets.FINNHUB_API_KEY }}
        run: python fetch_prices.py

      # Backfills a year of history the first time it runs, then appends each
      # day's close. Also writes market.json with the S&P, Nasdaq, VIX, Brent
      # and 10-year Treasury levels for the homepage. Never fails the job: if
      # the source is unreachable, the site keeps the numbers it already has.
      - name: Update price history and market levels
        run: python fetch_history.py || echo "History/market update failed — keeping existing data."

      - name: Rebuild the site data
        run: python build.py

      - name: Commit the updated data
        run: |
          git config user.name  "freeflow-bot"
          git config user.email "actions@github.com"
          git add prices.json docs/data.js || true
          git add docs/history.js market.json || true
          # git commit exits non-zero when nothing changed (e.g. a holiday),
          # which is normal and shouldn't fail the run.
          if git diff --staged --quiet; then
            echo "No price changes today — nothing to commit."
          else
            git commit -m "Update prices $(date -u +'%Y-%m-%d')"
            git push
            echo "Pushed updated prices."
          fi
